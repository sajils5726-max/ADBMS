CREATE OR REPLACE PROCEDURE PR2
(
 P_ENO IN EMPLOY.ENO%TYPE,
 P_COMM OUT NUMBER
)
AS
BEGIN
 SELECT MAX(SAL)*0.10
 INTO P_COMM
 FROM EMPLOY
 WHERE ENO=P_ENO;
END;


DECLARE
N EMPLOY.ENO%TYPE:=&N;
COMM NUMBER;
BEGIN
PR2(N,COMM);
DBMS_OUTPUT.PUT_LINE('COMMISSION IS'||COMM);
END;
